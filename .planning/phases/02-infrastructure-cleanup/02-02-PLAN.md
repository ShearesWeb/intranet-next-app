<objective>
Extract shared managed CCA business logic into a lib function and replace all self-calling API routes with direct function calls.

Purpose: Three API routes (getCCAs, rankings/all, managedPositions) currently call the getManagedCCAs route via internal HTTP fetch with manual cookie forwarding. This creates unnecessary latency, fragile cookie forwarding, and URL resolution issues across environments. Per D-04, D-05, and D-06, the business logic is extracted to `src/lib/managed-cca-data.ts` while both routes remain as separate API endpoints.

Output: New `src/lib/managed-cca-data.ts` with `getManagedCcaData()`, 3 routes refactored to use direct calls, getManagedCCAs becomes thin wrapper, unit test for extracted function.
</objective>

import { createClient } from "@/utils/supabase/server";
import { fetchUserInfo } from "@/lib/fetch-user-info";
import { getManagedCcaData } from "@/lib/managed-cca-data";

// Re-export types for backward compatibility with any direct importers
export type { ManagedCcaPosition, ManagedCcaPositionsRow } from "@/lib/managed-cca-data";

export async function GET() {
    try {
        const user = await fetchUserInfo();
        if (!user) {
            return NextResponse.json({ error: "Unauthorized" }, { status: 401 });
        }

        const supabase = await createClient();
        const data = await getManagedCcaData(supabase, user);
        return NextResponse.json({ data });
    } catch (error: unknown) {
        const message = error instanceof Error ? error.message : error;
        console.error("Error in getManagedCCAs:", message);
        return NextResponse.json({ error: "Failed to fetch managed CCAs" }, { status: 500 });
    }
}
