   Replace ALL remaining inline JSX with component composition. The final page should look approximately like:

   ```typescript
   "use client";

   import React, { useState } from "react";
   import { useSearchParams } from "next/navigation";
   import { AppLayout } from "@/components/layout/app-layout";
   import { LoadingSpinner } from "@/components/ui/loading-spinner";
   import { Alert, AlertDescription, AlertTitle } from "@/components/ui/alert";
   import { Button } from "@/components/ui/button";
   import { AlertCircleIcon, ArrowLeft, Volleyball } from "lucide-react";
   import Link from "next/link";

   import { useCcaMembers } from "@/features/cca/hooks/use-cca-members";
   import { MemberGroupTable } from "@/features/cca/components/member-group-table";
   import { AddMemberModal } from "@/features/cca/components/add-member-modal";
   import { SaveErrorDialog } from "@/features/cca/components/save-error-dialog";
   import { isChairPositionType } from "@/features/cca/utils/member-helpers";

   export default function ManageMembersPage() {
     const searchParams = useSearchParams();
     const ccaId = searchParams.get("ccaId");

     const {
       groupedMembers, cca, permissions, memberPositionId, allPositions,
       loading, error, saving, saveErrors, setSaveErrors,
       pendingChanges,
       searchQuery, setSearchQuery, searchResults, searchLoading, searchUser,
       addMember, removeMember, updateStatus, updateCommitment,
       saveAll,
     } = useCcaMembers(ccaId);

     const [showAddMemberModal, setShowAddMemberModal] = useState(false);

     const isSportsCulture = permissions.ccaType === "sports" || permissions.ccaType === "culture";

     if (loading) return <AppLayout><LoadingSpinner /></AppLayout>;
     if (error) return <AppLayout><Alert variant="destructive">...</Alert></AppLayout>;

     // Determine group order for rendering
     const groupOrder = Object.keys(groupedMembers);

     return (
       <AppLayout>
         {/* Header with back link, CCA name, member counts */}
         {/* Pending changes bar with save button */}

         {groupOrder.map((groupName) => (
           <MemberGroupTable
             key={groupName}
             title={groupName}
             members={groupedMembers[groupName]}
             canManage={permissions.canManageMembers}
             ccaType={permissions.ccaType}
             onRemoveMember={removeMember}
             onUpdateStatus={updateStatus}
             onUpdateCommitment={updateCommitment}
             onAddMember={() => setShowAddMemberModal(true)}
             showActions={!isChairPositionType(groupName)}
             showAddButton={groupName === "member" && isSportsCulture}
           />
         ))}

         <AddMemberModal
           open={showAddMemberModal}
           onOpenChange={setShowAddMemberModal}
           searchQuery={searchQuery}
           onSearchQueryChange={setSearchQuery}
           searchResults={searchResults}
           searchLoading={searchLoading}
           onSearch={searchUser}
           onAddMember={addMember}
           allPositions={allPositions}
           ccaType={permissions.ccaType}
           isSportsCulture={isSportsCulture}
         />

         <SaveErrorDialog
           errors={saveErrors}
           onClose={() => setSaveErrors([])}
         />

         {saving && <LoadingSpinner message="Saving changes..." overlay />}
       </AppLayout>
     );
   }
   ```
